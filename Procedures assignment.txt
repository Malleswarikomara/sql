15. DELIMITER //

CREATE PROCEDURE IncreaseSalary(
    IN Eid INT,
    IN percentage DECIMAL(5,2),
    INOUT esal DECIMAL(10,2)
)
BEGIN
    SELECT salary INTO esal FROM employee
    WHERE emp_id = Eid;
    SET esal = esal + (esal * percentage / 100);
    UPDATE employee
    SET salary = esal
    WHERE emp_id = Eid;
   END //

   DELIMITER ; 
  
   SET @sal = 0;

  CALL IncreaseSalary(101, 10, @sal);

  SELECT @sal;

17. DELIMITER //

CREATE PROCEDURE GetHighestPaidEmployee(
    IN dept_name VARCHAR(30))
BEGIN
    SELECT *
    FROM employee
    WHERE department = dept_name
    ORDER BY salary DESC
    LIMIT 1;
END //

DELIMITER ;
  
  call GetHighestPaidEmployee('IT');

18.  DELIMITER //

CREATE PROCEDURE GetLowestPaidEmployee(
    IN dept_name VARCHAR(30)
)
BEGIN
    SELECT *
    FROM employee
    WHERE department = dept_name
    ORDER BY salary ASC
    LIMIT 1;
END //

DELIMITER ;

call GetLowestPaidEmployee('IT');
